ANDROID/RELAY-SERVER.MJS:

#!/data/data/com.termux/files/usr/bin/node
import http from 'http';
import { exec } from 'child_process';
import fs from 'fs';

const PORT = process.env.RELAY_PORT || 8080;
const API_KEY = process.env.RELAY_API_KEY || 'change-me';

// Allowed commands (security whitelist)
const ALLOWED_COMMANDS = [
  'termux-sms-send',
  'termux-camera-photo',
  'termux-location',
  'termux-battery-status',
  'ls', 'pwd', 'cat', 'echo'
];

const server = http.createServer((req, res) => {
  if (req.method === 'POST' && req.url === '/execute') {
    let body = '';
    req.on('data', chunk => body += chunk);
    req.on('end', () => {
      const { apiKey, command, args } = JSON.parse(body);

      // Verify API key
      if (apiKey !== API_KEY) {
        res.writeHead(401);
        return res.end(JSON.stringify({ error: 'Invalid API key' }));
      }

      // Check whitelist
      const baseCmd = command.split(' ')[0];
      if (!ALLOWED_COMMANDS.includes(baseCmd)) {
        res.writeHead(403);
        return res.end(JSON.stringify({ error: 'Command not allowed' }));
      }

      // Execute
      const fullCmd = args ? `${command} ${args.join(' ')}` : command;
      exec(fullCmd, { timeout: 30000 }, (error, stdout, stderr) => {
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify({
          success: !error,
          stdout,
          stderr,
          error: error?.message
        }));
      });
    });
  } else {
    res.writeHead(404);
    res.end('Not found');
  }
});

server.listen(PORT, () => {
  console.log(`🤖 Termux Relay running on port ${PORT}`);
  console.log(`📱 Phone is now a compute node`);
});

#!/data/data/com.termux/files/usr/bin/bash

echo "🤖 Setting up Termux Relay..."

pkg update -y
pkg upgrade -y

# Install Node.js
pkg install -y nodejs

# Install Termux API
pkg install -y termux-api

# Create config
mkdir -p ~/.termux-relay
cat > ~/.termux-relay/config.json <<EOF
{
  "port": 8080,
  "apiKey": "$(openssl rand -hex 16)",
  "allowedCommands": [
    "termux-sms-send",
    "termux-camera-photo",
    "termux-location",
    "termux-battery-status"
  ]
}
EOF

echo "✅ Setup complete!"
echo "API Key: $(cat ~/.termux-relay/config.json | grep apiKey)"
echo ""
echo "Start relay: node relay-server.mjs"

CLIENT/RELAY-CLIENT.MJS:

import http from 'http';

export class TermuxRelay {
  constructor(phoneUrl, apiKey) {
    this.url = new URL(phoneUrl);
    this.apiKey = apiKey;
  }

  async execute(command, args = []) {
    const data = JSON.stringify({ apiKey: this.apiKey, command, args });
    const options = {
      hostname: this.url.hostname,
      port: this.url.port,
      path: '/execute',
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Content-Length': data.length
      }
    };

    return new Promise((resolve, reject) => {
      const req = http.request(options, res => {
        let body = '';
        res.on('data', chunk => body += chunk);
        res.on('end', () => resolve(JSON.parse(body)));
      });
      req.on('error', reject);
      req.write(data);
      req.end();
    });
  }

  async sendSMS(number, message) {
    return this.execute('termux-sms-send', ['-n', number, message]);
  }

  async takePhoto(camera = 0) {
    return this.execute('termux-camera-photo', ['-c', camera, '/sdcard/photo.jpg']);
  }

  async getLocation() {
    return this.execute('termux-location');
  }

  async getBattery() {
    return this.execute('termux-battery-status');
  }
}
