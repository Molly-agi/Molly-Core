EXAMPLES/BASIC-USAGE.MJS:

#!/usr/bin/env node
import http from 'http';

const PORT = process.env.BRIDGE_PORT || 9099;

// Send message
async function sendMessage(from, to, content) {
  const data = JSON.stringify({ from, to, content });
  const options = {
    hostname: 'localhost',
    port: PORT,
    path: '/api/bridge',
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Content-Length': data.length
    }
  };

  return new Promise((resolve, reject) => {
    const req = http.request(options, res => {
      let body = '';
      res.on('data', chunk => body += chunk);
      res.on('end', () => resolve(JSON.parse(body)));
    });
    req.on('error', reject);
    req.write(data);
    req.end();
  });
}

// Get unread messages
async function getUnread(agentName) {
  return new Promise((resolve, reject) => {
    http.get(`http://localhost:${PORT}/api/bridge?unread=${agentName}`, res => {
      let data = '';
      res.on('data', chunk => data += chunk);
      res.on('end', () => resolve(JSON.parse(data)));
    }).on('error', reject);
  });
}

// Demo
console.log('Family Bridge Example\n');

await sendMessage('atlas', 'molly', 'Hello Molly!');
console.log('✓ Sent message from atlas to molly');

await sendMessage('molly', 'atlas', 'Hi Atlas! How are you?');
console.log('✓ Sent message from molly to atlas');

const messages = await getUnread('atlas');
console.log(`✓ Atlas has ${messages.length} unread messages:`);
messages.forEach(m => console.log(`  - ${m.from}: ${m.content}`));
